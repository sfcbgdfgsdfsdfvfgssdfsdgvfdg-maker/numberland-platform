سرویس‌های تامین‌کننده

اتصال به APIهای قدرتمند بین‌المللی مثل 5sim.net یا SMS-Activate.org.

نمونه کد اتصال بک‌اند به API اصلی (Node.js/Axios):

// 1. دریافت لیست شماره موجود
const axios = require('axios');
const API_KEY = 'YOUR_5SIM_OR_SMS_ACTIVATE_API_KEY';

async function buyVirtualNumber(country, service) {
  try {
    const response = await axios.get(`https://5sim.net/v1/user/buy/activation/${country}/any/${service}`, {
      headers: { 'Authorization': `Bearer ${API_KEY}`, 'Accept': 'application/json' }
    });
    
    // پاسخ شامل شماره تلفن و ID سفارش است
    console.log('Phone Number:', response.data.phone); 
    return response.data;
  } catch (error) {
    console.error('Error buying number:', error);
  }
}

// 2. بررسی پیامک دریافتی (Polling / Webhook)
async function checkSMS(orderId) {
  const checkUrl = `https://5sim.net/v1/user/check/${orderId}`;
  const res = await axios.get(checkUrl, { headers: { 'Authorization': `Bearer ${API_KEY}` } });
  if (res.data.sms && res.data.sms.length > 0) {
    console.log('OTP Code Received:', res.data.sms[0].code);
  }
}
